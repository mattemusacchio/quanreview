%  (b) Included:    preamble needs  \usepackage{tikz}
%                   body:
%                       \begin{figure*}[t]\centering

%                         \caption{...}\label{fig:architecture}
%                       \end{figure*}
%  The guard detects the mode by testing whether TikZ is already loaded.
%
%  GEOMETRY
%  --------
%  Coordinates are in abstract units; 1 unit = \AFU cm is set so the
%  drawing is exactly 13.50 units = 6.6in wide (full ACL text width).
%  Canvas 13.50 x 9.95 units = 16.76 x 12.36 cm = 6.60 x 4.87 in,
%  aspect 1.357:1.  At that size 1 unit = 35.20pt, so the smallest text
%  in the figure (0.205 units) sets at 7.22pt.  KEEP EVERY FONT AT OR
%  ABOVE 0.205 units if you change \AFU or the canvas width.
%
%  LAYOUT
%  ------
%  Four full-width horizontal bands, strict top-to-bottom flow,
%  alternating automated / human so the human-in-the-loop boundary is
%  unmistakable.  No edge crossings.
%    A  automated  (1) corpus + config      (2) select & assign
%    B  human      (3) blind pairwise review
%    C  automated  (4) outputs  (5) integration  (6) agreement report
%    D  human      (7) conflict adjudication
%
%  COLOUR
%  ------
%  Grayscale-safe by construction: all text is black, and the only
%  accent (AFblind) is separated from the other fills by LIGHTNESS
%  (white 1.00 > AFdata 0.91 > AFblind 0.82), not by hue.  Human bands
%  are additionally marked by hatching and a heavy dashed border.
%
%  RESTYLING
%  ---------
%  Change the \definecolor block and the \tikzset block; leave the body.
% =====================================================================

% \AFnode{style}{x}{y}{w}{h}{content}  -- (x,y) is the south-west corner,
% w/h in units.  Never put a \node (e.g. \AFbadge) inside #6.
\providecommand{\AFnode}[6]{%
  \pgfmathsetlengthmacro{\AFw}{#4*\AFU cm}%
  \pgfmathsetlengthmacro{\AFh}{#5*\AFU cm}%
  \pgfmathsetlengthmacro{\AFtw}{(#4-0.20)*\AFU cm}%
  \node[#1, minimum width=\AFw, minimum height=\AFh, text width=\AFtw]
       at (#2,#3) {#6};%
}
% \AFbadge{x}{y}{n} -- numbered disc centred on (x,y); 0.296u = 0.368cm
\providecommand{\AFbadge}[3]{%
  \node[circle, fill=AFink, text=white, inner sep=0pt, minimum size=0.368cm,
        font=\fontsize{7.22}{7.22}\selectfont\bfseries] at (#1,#2) {#3};%
}
% \AFband{y0}{y1}{style} -- fill + border only
\providecommand{\AFband}[3]{%
  \draw[#3] (0.15,#1) rectangle (13.35,#2);%
}
% \AFbandlabel{y1}{label} -- drawn after any hatching so text stays clean
\providecommand{\AFbandlabel}[2]{%
  \node[AFlane] at (13.15,#1-0.335) {#2};%
}
% \AFhatchband{y0}{y1} -- diagonal hatching for the human bands
\providecommand{\AFhatchband}[2]{%
  \begin{scope}
    \clip (0.15,#1) rectangle (13.35,#2);
    \pgfmathsetmacro{\AFspan}{#2-#1}%
    \foreach \AFk in {-4,-3.66,...,13.5}{%
      \draw[AFhatch, line width=0.35pt] (\AFk,#1) -- (\AFk+\AFspan,#2);}
  \end{scope}%
}
